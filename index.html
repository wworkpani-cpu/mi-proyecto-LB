// Pega este código en Extensiones > Apps Script de tu hoja de Google. var HOJA_PRODUCTOS = "Productos"; var HOJA_VENTAS = "Ventas"; function hoja_(nombre, titulos) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(nombre); if (!sh) { sh = ss.insertSheet(nombre); sh.appendRow(titulos); sh.setFrozenRows(1); sh.getRange(1, 1, 1, titulos.length).setFontWeight("bold").setBackground("#f5b700"); if (nombre === HOJA_VENTAS) sh.getRange("A:A").setNumberFormat("@"); } return sh; } function productos_() { return hoja_(HOJA_PRODUCTOS, ["ID", "Producto", "Precio", "Existencia", "Mínimo"]); } function cotizaciones_() { return hoja_("Cotizaciones", ["Número", "Fecha", "Cliente", "Total", "Datos"]); } function ventas_() { return hoja_(HOJA_VENTAS, ["Fecha", "Detalle", "Total", "Items"]); } function salida_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } // Lee todo: se usa cuando se abre el sistema. function doGet() { try { var p = productos_().getDataRange().getValues().slice(1) .filter(function (r) { return r[0] !== ""; }) .map(function (r) { return { id: +r[0], n: String(r[1]), p: +r[2], s: +r[3], m: +r[4] }; }); var s = ventas_().getDataRange().getValues().slice(1) .filter(function (r) { return r[0] !== ""; }) .map(function (r) { return { d: (r[0] instanceof Date) ? r[0].toISOString() : String(r[0]), t: +r[2], items: JSON.parse(r[3] || "[]") }; }); var q = cotizaciones_().getDataRange().getValues().slice(1) .filter(function (r) { return r[0] !== ""; }) .map(function (r) { return JSON.parse(r[4] || "{}"); }); return salida_({ ok: true, p: p, s: s, q: q }); } catch (err) { return salida_({ ok: false, error: String(err) }); } } // Guarda: productos, una venta nueva o borrar ventas. function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var b = JSON.parse(e.postData.contents); if (b.p) { var sp = productos_(); var ultima = sp.getLastRow(); if (ultima > 1) sp.getRange(2, 1, ultima - 1, 5).clearContent(); if (b.p.length) { sp.getRange(2, 1, b.p.length, 5).setValues( b.p.map(function (x) { return [x.id, x.n, x.p, x.s, x.m]; }) ); } } if (b.clearSales) { var sv = ventas_(); var u = sv.getLastRow(); if (u > 1) sv.getRange(2, 1, u - 1, 4).clearContent(); } if (b.sale) { var detalle = b.sale.items.map(function (i) { return i.q + "x " + i.n; }).join("; "); ventas_().appendRow([b.sale.d, detalle, b.sale.t, JSON.stringify(b.sale.items)]); } if (b.quote) { var sc = cotizaciones_(); var vals = sc.getDataRange().getValues(); var fila = -1; for (var i = 1; i < vals.length; i++) { if (String(vals[i][0]) === String(b.quote.n)) { fila = i + 1; break; } } var reg = [b.quote.n, b.quote.d, b.quote.cli, b.quote.t, JSON.stringify(b.quote)]; if (fila > 0) sc.getRange(fila, 1, 1, 5).setValues([reg]); else sc.appendRow(reg); } return salida_({ ok: true }); } catch (err) { return salida_({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } }
